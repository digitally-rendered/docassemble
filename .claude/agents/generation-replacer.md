@dataclass
class YAMLSection:
    """Clean YAML section representation"""
    section_type: str  # metadata, include, objects, question, code, attachment
    content: Dict[str, Any] = field(default_factory=dict)
    
    def to_yaml_dict(self) -> Dict[str, Any]:
        """Convert to clean dictionary for YAML serialization"""
        return self.content

@dataclass  
class InterviewDocument:
    """Complete interview document using YAML objects"""
    sections: List[YAMLSection] = field(default_factory=list)
    
    def add_metadata_section(self, title: str, form_number: str):
        """Add metadata section as YAML object"""
        metadata_section = YAMLSection(
            section_type="metadata",
            content={
                "metadata": {
                    "title": f"Ontario Form {form_number} - {title}",
                    "short title": f"Form {form_number}",
                    "version": "2.0",
                    "author": "Enhanced Form Generator"
                }
            }
        )
        self.sections.append(metadata_section)
        
    def add_includes_section(self, includes: List[str]):
        """Add includes as YAML object"""
        include_section = YAMLSection(
            section_type="include",
            content={"include": includes}
        )
        self.sections.append(include_section)
        
    def add_objects_section(self, objects: List[Dict[str, str]]):
        """Add objects section as YAML object"""
        objects_section = YAMLSection(
            section_type="objects", 
            content={"objects": objects}
        )
        self.sections.append(objects_section)
        
    def add_question_block(self, question_data: Dict[str, Any]):
        """Add question block as clean YAML object"""
        question_section = YAMLSection(
            section_type="question",
            content=question_data
        )
        self.sections.append(question_section)
        
    def add_code_block(self, code_content: str, mandatory: bool = False):
        """Add code block as YAML object"""
        code_data = {"code": code_content}
        if mandatory:
            code_data["mandatory"] = True
            
        code_section = YAMLSection(
            section_type="code",
            content=code_data
        )
        self.sections.append(code_section)
        
    def to_yaml(self) -> str:
        """Convert entire document to YAML using proper encoding"""
        yaml_parts = []
        
        for section in self.sections:
            yaml_parts.append("---")
            # Use yaml.dump() with proper settings, NOT string concatenation
            yaml_content = yaml.dump(
                section.to_yaml_dict(),
                default_flow_style=False,
                allow_unicode=True,
                sort_keys=False,
                width=1000  # Prevent unwanted line breaks
            )
            yaml_parts.append(yaml_content)
            
        return '\n'.join(yaml_parts)

                                               form_config: FormConfiguration) -> str:
        """Generate interview using existing parsed data + new YAML approach"""
        
        interview = InterviewDocument()
        
        # Add metadata
        interview.add_metadata_section(form_config.form_title, form_number)
        
        # Add standard includes
        interview.add_includes_section([
            'shared/person-fields.yml',
            'shared/court-fields.yml',
            'shared/validation-functions.yml'
        ])
        
        # Add objects based on form type
        interview.add_objects_section(self._generate_objects_for_form(form_config))
        
        # Add intro question
        interview.add_question_block(self._create_intro_question(form_config))
        
        # Process validated fields into questions
        for field in validated_fields:
            if self._should_create_question_for_field(field):
                question_data = self._create_question_from_validated_field(field)
                interview.add_question_block(question_data)
                
        # Add review screen
        interview.add_question_block(self._create_review_screen(validated_fields))
        
        # Add document assembly
        interview.add_code_block(self._create_document_assembly_code(form_number))
        
        return interview.to_yaml()
        
    def _create_question_from_validated_field(self, field: ValidatedField) -> Dict[str, Any]:
        """Convert ValidatedField to clean question YAML object"""
        
        question_data = {
            "question": field.field_label or f"Please provide {field.canonical_name}",
            "fields": []
        }
        
        # Build field definition as data structure
        field_def = {field.field_label: field.docassemble_path}
        
        # Add datatype based on field type
        if field.field_type == 'date':
            field_def["datatype"] = "date"
        elif field.field_type == 'currency':
            field_def["datatype"] = "currency"
        elif field.field_type == 'number':
            field_def["datatype"] = "number"
        elif field.field_type == 'email':
            field_def["datatype"] = "email"
            
        question_data["fields"].append(field_def)
        
        # Add validation from existing field validation rules
        if field.validation_rules:
            validation_code = self._convert_validation_rules_to_code(field.validation_rules)
            if validation_code:
                question_data["validation code"] = validation_code
                
        # Add help text if available
        if hasattr(field, 'help_text') and field.help_text:
            question_data["help"] = field.help_text
            
        return question_data
        
    def _generate_objects_for_form(self, form_config: FormConfiguration) -> List[Dict[str, str]]:

            objects.extend([
                {"DADict": "assets"},
                {"DADict": "debts"}, 
                {"DADict": "income"}
            ])

        self.demo_base_path = '/docassemble_demo/docassemble/demo/data/questions/examples'

        """Extract proven patterns from demo library"""
        
        patterns = {}
        
        # Load and parse actual demo files
        patterns['signature_pattern'] = self._load_pattern('multi-signature.yml')
        patterns['object_collection'] = self._load_pattern('object-checkboxes.yml')
        patterns['conditional_logic'] = self._load_pattern('branch-mandatory.yml')
        patterns['validation_examples'] = self._extract_validation_patterns()
        
        return patterns
        
    def apply_pattern_to_interview(self, interview: InterviewDocument, 
                                 pattern_name: str, form_config: FormConfiguration):
        """Apply demo pattern to interview document"""
        
        patterns = self.extract_working_patterns()
        pattern = patterns.get(pattern_name)
        
        if pattern_name == 'signature_pattern' and form_config.requires_signatures:
            signature_question = self._adapt_signature_pattern(pattern, form_config)
            interview.add_question_block(signature_question)
            
        elif pattern_name == 'object_collection' and form_config.has_children_section:
            children_questions = self._adapt_collection_pattern(pattern, 'children')
            for question in children_questions:
                interview.add_question_block(question)
                
    def _load_pattern(self, filename: str) -> Dict[str, Any]:
        """Load and parse demo YAML file"""
        
        pattern_path = os.path.join(self.demo_base_path, filename)
        
        with open(pattern_path, 'r') as f:
            # Parse actual YAML content
            pattern_blocks = list(yaml.safe_load_all(f))
            
        return {
            'blocks': pattern_blocks,
            'metadata': self._extract_pattern_metadata(pattern_blocks)
        }

class NewGenerationOrchestrator:
    """Orchestrate new generation system using existing parsing"""

        from automated_form_processor import AutomatedFormProcessor
        from field_validation_mapper import FieldValidationMapper

        # New generation components

        self.demo_integrator = DemoPatternIntegrator()
        self.error_integrator = ErrorHandlerIntegrator()
        
    def generate_all_forms_with_new_system(self):
        """Generate all forms using existing parsing + new generation"""
        
        print("🔄 Using existing parsing, new generation system...")
        
        # Load existing forms registry (keep this!)
        forms_registry = self.form_processor.load_forms_registry()
        
        for form_number, form_config in forms_registry.items():
            print(f"📝 Generating Form {form_number} with new system...")
            
            # Use existing parsed and validated fields (don't re-parse!)
            validated_fields = self._load_existing_validated_fields(form_number)
            
            # Generate with new system
            interview_yaml = self.interview_generator.generate_interview_from_validated_fields(
                form_number, validated_fields, form_config
            )
            
            # Create interview document object
            interview = self._yaml_to_interview_object(interview_yaml)
            
            # Apply demo patterns
            self.demo_integrator.apply_pattern_to_interview(interview, 'signature_pattern', form_config)
            if form_config.has_children_section:
                self.demo_integrator.apply_pattern_to_interview(interview, 'object_collection', form_config)
                
            # Add error handling
            self.error_integrator.add_error_handling_to_interview(interview)
            
            # Final YAML output
            final_yaml = interview.to_yaml()
            
            # Write output
            output_path = f"output/interviews/form_{form_number}_new.yml"
            with open(output_path, 'w') as f:
                f.write(final_yaml)
                
            print(f"✅ Form {form_number} generated with new system")
            
    def _load_existing_validated_fields(self, form_number: str) -> List[ValidatedField]:
        """Load existing validated fields (reuse all parsing work!)"""
        
        # Use existing parsed JSON files
        parsed_file = f"utilities/parsed_forms/form_{form_number}_fields.json"

        # Convert to ValidatedField objects using existing mapper
        validated_fields = self.field_mapper.convert_parsed_to_validated_fields(field_data)
