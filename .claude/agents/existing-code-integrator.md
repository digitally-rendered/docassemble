class ExistingParserWrapper:
    """Wrapper around existing parsing infrastructure"""
    
    def __init__(self):
        # Initialize your existing parsers
        self.automated_processor = AutomatedFormProcessor()
        self.comprehensive_parser = ComprehensiveFormParser()
        self.validation_mapper = FieldValidationMapper()
        self.hybrid_parser = HybridFormParser()
        
    def get_parsed_form_data(self, form_number: str) -> Dict[str, Any]:
        """Get parsed data using existing infrastructure"""
        
        # Use your existing parsed_forms JSON files
        parsed_file = f"parsed_forms/form_{form_number}_fields.json"
        
        if os.path.exists(parsed_file):
            with open(parsed_file, 'r') as f:
                return json.load(f)
        else:
            # Fall back to parsing if not cached
            return self._parse_form_fresh(form_number)
            
    def get_validated_fields(self, form_number: str) -> List[ValidatedField]:
        """Get validated fields using existing FieldValidationMapper"""
        
        raw_fields = self.get_parsed_form_data(form_number)
        validated_fields = self.validation_mapper.validate_and_deduplicate_fields(
            raw_fields, form_number
        )
        
        return validated_fields
        
    def get_form_configuration(self, form_number: str) -> FormConfiguration:
        """Get form config using existing FormRegistry"""
        
        return self.automated_processor.get_form_configuration(form_number)
        
    def _parse_form_fresh(self, form_number: str) -> Dict[str, Any]:
        """Parse form using existing parsing pipeline if not cached"""
        
        # Use existing hybrid parsing approach
        return self.hybrid_parser.parse_form_comprehensive(form_number)

from field_validation_mapper import ValidatedField, FieldValidationMapper

class EnhancedFieldMapper:
    """Enhance existing field mapping with YAML object generation"""
    
    def __init__(self):
        self.existing_mapper = FieldValidationMapper()
        
    def map_validated_fields_to_yaml_objects(self, 
                                           validated_fields: List[ValidatedField]
                                           ) -> List[QuestionBlock]:
        """Convert ValidatedField objects to QuestionBlock YAML objects"""
        
        question_blocks = []
        
        for field in validated_fields:
            # Use existing field metadata
            question_block = QuestionBlock()
            
            question_block.question = field.field_label
            question_block.fields = [{
                field.field_label: field.docassemble_path
            }]
            
            # Use existing validation rules
            if field.validation_rules:
                question_block.validation_code = self._convert_validation_rules(
                    field.validation_rules
                )
                
            # Use existing field type mapping
            if field.field_type == 'date':
                question_block.fields[0]['datatype'] = 'date'
            elif field.field_type == 'currency':
                question_block.fields[0]['datatype'] = 'currency'
                
            question_blocks.append(question_block)
            
        return question_blocks
        
    def get_shared_field_mappings(self) -> Dict[str, Any]:
        """Use existing field validation to identify shared fields"""
        
        # Your existing code already does field deduplication
        shared_fields = self.existing_mapper.get_common_fields_across_forms()
        
        # Convert to the format needed for shared modules
        shared_mappings = {}
        for field_group, fields in shared_fields.items():
            shared_mappings[field_group] = {
                'fields': [field.docassemble_path for field in fields],
                'validation': [field.validation_rules for field in fields],
                'reuse_percentage': self._calculate_reuse_percentage(fields)
            }
            
        return shared_mappings
