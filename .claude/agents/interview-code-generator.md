```python  
@dataclass
class InterviewDocument:
    """Clean interview document structure using YAML objects"""
    sections: List[Dict[str, Any]] = field(default_factory=list)
    
    def add_metadata_section(self, title: str, form_number: str):
        """Add metadata section as YAML object"""
        self.sections.append({
            "metadata": {
                "title": f"Ontario Form {form_number} - {title}",
                "short title": f"Form {form_number}",
                "version": "2.0",
                "author": "Enhanced Form Generator"
            }
        })
        
    def add_includes_section(self, includes: List[str]):
        """Add includes as YAML object"""
        self.sections.append({"include": includes})
        
    def add_objects_section(self, objects: List[Dict[str, str]]):
        """Add objects section as YAML object"""
        self.sections.append({"objects": objects})
        
    def add_question_block(self, question_data: Dict[str, Any]):
        """Add question block as clean YAML object"""
        self.sections.append(question_data)
        
    def to_yaml(self) -> str:
        """Convert entire document to YAML using proper encoding"""
        yaml_parts = []
        
        for section in self.sections:
            yaml_parts.append("---")
            # Use yaml.dump() with proper settings, NOT string concatenation
            yaml_content = yaml.dump(
                section,
                default_flow_style=False,
                allow_unicode=True,
                sort_keys=False,
                width=1000  # Prevent unwanted line breaks
            )
            yaml_parts.append(yaml_content)
            
        return '\n'.join(yaml_parts)

class NewInterviewGenerator:
    """Brand new interview generator replacing old system"""
    
    def __init__(self):
        # Keep using existing parsing infrastructure
        self.field_mapper = FieldValidationMapper()
        self.demo_patterns = DemoPatternExtractor()
        
    def generate_interview_from_validated_fields(self, 
                                               form_number: str,
                                               validated_fields: List[ValidatedField],
                                               form_config: FormConfiguration) -> str:
        """Generate interview using existing parsed data + new YAML approach"""
        
        interview = InterviewDocument()
        
        # Add metadata
        interview.add_metadata_section(form_config.form_title, form_number)
        
        # Add standard includes
        interview.add_includes_section([
            'shared/person-fields.yml',
            'shared/court-fields.yml',
            'shared/validation-functions.yml'
        ])
        
        # Add objects based on form type
        interview.add_objects_section(self._generate_objects_for_form(form_config))
        
        # Add intro question
        interview.add_question_block(self._create_intro_question(form_config))
        
        # Process validated fields into questions
        for field in validated_fields:
            if self._should_create_question_for_field(field):
                question_data = self._create_question_from_validated_field(field)
                interview.add_question_block(question_data)
                
        # Add review screen
        interview.add_question_block(self._create_review_screen(validated_fields))
        
        # Add document assembly
        interview.add_question_block({
            "code": self._create_document_assembly_code(form_number),
            "mandatory": True
        })
        
        return interview.to_yaml()
        
    def _create_question_from_validated_field(self, field: ValidatedField) -> Dict[str, Any]:
        """Convert ValidatedField to clean question YAML object"""
        
        question_data = {
            "question": field.field_label or f"Please provide {field.canonical_name}",
            "fields": []
        }
        
        # Build field definition as data structure
        field_def = {field.field_label: field.docassemble_path}
        
        # Add datatype based on field type
        if field.field_type == 'date':
            field_def["datatype"] = "date"
        elif field.field_type == 'currency':
            field_def["datatype"] = "currency"
        elif field.field_type == 'number':
            field_def["datatype"] = "number"
        elif field.field_type == 'email':
            field_def["datatype"] = "email"
            
        question_data["fields"].append(field_def)
        
        # Add validation from existing field validation rules
        if field.validation_rules:
            validation_code = self._convert_validation_rules_to_code(field.validation_rules)
            if validation_code:
                question_data["validation code"] = validation_code
                
        # Add help text if available
        if hasattr(field, 'help_text') and field.help_text:
            question_data["help"] = field.help_text
            
        return question_data
```

```python
            if 'question' in block and 'fields' in block:

                    'question_template': block['question'],
                    'fields_structure': block['fields']
                }
                
        return patterns
```

```python
class FieldMapper:
    def __init__(self):
        self.shared_patterns = self._load_shared_patterns()
        self.validation_library = self._load_validation_library()
        
    def map_field_to_docassemble(self, field_data):
        """Convert parsed field data to Docassemble question"""
        field_type = field_data.get('type')
        
        if field_type == 'text':
            return self._generate_text_field(field_data)
        elif field_type == 'date':
            return self._generate_date_field(field_data) 
        elif field_type == 'checkbox':
            return self._generate_checkbox_field(field_data)
        elif field_type == 'select':
            return self._generate_select_field(field_data)
            
    def _generate_text_field(self, field_data):
        # Apply demo patterns + error handling
        return {
            'question': field_data.get('label'),
            'fields': [{
                field_data.get('docassemble_name'): field_data.get('variable')
            }],
            'validation_code': self._generate_validation_code(field_data),
            'help': self._generate_help_text(field_data)
        }
        
    def _generate_validation_code(self, field_data):
        # Use error handler patterns from common intake
        validations = []
        
        if field_data.get('required'):
            validations.append(f"if not {field_data.get('variable')}:")
            validations.append('  validation_error("This field is required")')
            
        if field_data.get('format') == 'postal_code':
            validations.append(self.validation_library.postal_code_validation())
            
        return '\n'.join(validations) if validations else None
```
