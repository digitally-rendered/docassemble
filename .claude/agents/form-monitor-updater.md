from typing import Dict, List, Tuple
import json
from deepdiff import DeepDiff

class FormChangeDetector:
    """Detect and analyze changes in form structures"""
    
    def __init__(self):
        self.field_analyzer = FieldAnalyzer()
        self.impact_assessor = ImpactAssessor()
        
    def analyze_form_changes(self, form_number: str, 
                           old_parsed_data: Dict, 
                           new_parsed_data: Dict) -> ChangeReport:
        """Analyze what changed between form versions"""
        
        changes = DeepDiff(old_parsed_data, new_parsed_data, 
                          ignore_order=True, 
                          verbose_level=2)
        
        change_report = ChangeReport(
            form_number=form_number,
            change_summary=self._categorize_changes(changes),
            impact_assessment=self._assess_impact(changes),
            regeneration_required=self._requires_regeneration(changes)
        )
        
        return change_report
        
    def _categorize_changes(self, changes: DeepDiff) -> Dict[str, List]:
        """Categorize the types of changes detected"""
        categories = {
            'fields_added': [],
            'fields_removed': [], 
            'fields_modified': [],
            'validation_changed': [],
            'labels_changed': []
        }
        
        # Analyze DeepDiff results and categorize
        if 'dictionary_item_added' in changes:
            categories['fields_added'] = list(changes['dictionary_item_added'])
            
        if 'dictionary_item_removed' in changes:
            categories['fields_removed'] = list(changes['dictionary_item_removed'])
            
        if 'values_changed' in changes:
            categories['fields_modified'] = list(changes['values_changed'])
            
        return categories
        
    def _assess_impact(self, changes: DeepDiff) -> ImpactAssessment:
        """Assess the impact of changes on existing interviews"""
        impact = ImpactAssessment()
        
        # Critical changes that break existing interviews
        if changes.get('dictionary_item_removed'):
            impact.severity = 'CRITICAL'
            impact.breaking_changes = True
            
        # Moderate changes that need interview updates  
        elif changes.get('dictionary_item_added'):
            impact.severity = 'MODERATE'
            impact.requires_interview_update = True
            
        # Minor changes (labels, help text, etc.)
        else:
            impact.severity = 'MINOR'
            impact.requires_regeneration = True
            
        return impact

@dataclass
class ChangeReport:
    """Report of changes detected in a form"""
    form_number: str
    change_summary: Dict[str, List]
    impact_assessment: 'ImpactAssessment'
    regeneration_required: bool
    timestamp: datetime = field(default_factory=datetime.now)
    
@dataclass 
class ImpactAssessment:
    """Assessment of change impact on interviews"""
    severity: str = 'UNKNOWN'  # CRITICAL, MODERATE, MINOR
    breaking_changes: bool = False
    requires_interview_update: bool = False
    requires_regeneration: bool = False

        self.pipeline_orchestrator = PipelineOrchestrator()
        self.test_runner = TestRunner()
        self.notification_service = NotificationService()
        
    def run_update_cycle(self):
        """Execute complete update cycle"""
        
        print("🔍 Checking for form updates...")
        updated_forms = self.form_monitor.check_all_forms()
        
        if not updated_forms:
            print("✅ No form updates detected")
            return
            
        print(f"📥 Found {len(updated_forms)} updated forms")
        
        for form_version in updated_forms:
            self._process_form_update(form_version)
            
    def _process_form_update(self, form_version: FormVersion):
        """Process a single form update"""
        
        print(f"🔄 Processing Form {form_version.form_number} update...")
        
        # 1. Parse the updated form
        parsed_data = self._parse_updated_form(form_version)
        
        # 2. Detect what changed
        old_data = self._load_previous_parsed_data(form_version.form_number)
        change_report = self.change_detector.analyze_form_changes(
            form_version.form_number, old_data, parsed_data
        )
        
        print(f"📊 Change analysis: {change_report.impact_assessment.severity}")
        
        # 3. Regenerate interview if needed
        if change_report.regeneration_required:
            success = self._regenerate_interview(form_version, parsed_data)
            
            if success:
                # 4. Run tests on updated interview
                test_results = self._run_tests(form_version.form_number)
                
                if test_results.all_passed:
                    # 5. Deploy updated interview
                    self._deploy_interview_update(form_version)
                    print(f"✅ Form {form_version.form_number} successfully updated")
                else:
                    self._handle_test_failures(form_version, test_results)
                    
        # 6. Notify stakeholders
        self._send_update_notification(form_version, change_report)
        
    def _regenerate_interview(self, form_version: FormVersion, 
                             parsed_data: Dict) -> bool:
        """Regenerate interview using existing parsing + enhanced generation"""
        
        try:
            # Use existing automated_form_processor for form handling
            from existing_parser_wrapper import ExistingParserWrapper
            from enhanced_generator_wrapper import EnhancedGeneratorWrapper
            
            parser_wrapper = ExistingParserWrapper()
            generator_wrapper = EnhancedGeneratorWrapper()

                form_version.form_number
            )

                form_version.form_number, parsed_data
            )
            
            # Write to file

            with open(output_path, 'w') as f:
                f.write(interview_yaml)
                
            return True
            
        except Exception as e:
            print(f"❌ Failed to regenerate Form {form_version.form_number}: {e}")
            return False
